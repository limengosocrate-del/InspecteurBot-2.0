(function(){
 'use strict';
 function init(){
  const workspace=document.getElementById('igt-workspace');
  const anchor=document.getElementById('igt-admin-stats');
  if(!workspace||!anchor||document.getElementById('igt-national-dashboard'))return;
  const panel=document.createElement('section');panel.id='igt-national-dashboard';
  panel.innerHTML='<div class="igt-nd-head"><div><h3>Tableau de bord national — ANI-IGT</h3><div class="igt-nd-sub">Vue de synthèse du registre actuellement intégré à l’annuaire. Les chiffres ne constituent pas encore un état certifié des effectifs en service.</div></div><button type="button" id="igt-nd-export">Exporter la synthèse CSV</button></div><div class="igt-nd-cards" id="igt-nd-cards"><div class="igt-nd-card"><small>Chargement</small><strong>—</strong></div></div><div class="igt-nd-grid"><div class="igt-nd-panel"><h4>Répartition par province</h4><div id="igt-nd-provinces">Préparation des données…</div></div><div class="igt-nd-panel"><h4>Qualité des informations</h4><div id="igt-nd-quality"></div></div></div><div class="igt-nd-panel" style="margin-top:12px"><h4>Répartition par catégorie et province</h4><div class="igt-nd-table-wrap" id="igt-nd-breakdown"></div></div><div class="igt-nd-note">Source : données embarquées dans la version actuelle de l’annuaire. Les agents administratifs et les agents issus du registre principal sont comptabilisés séparément pour éviter les doubles comptes. Une synchronisation certifiée avec la base Supabase et une validation des effectifs par les services compétents seront nécessaires avant usage comme statistique officielle.</div>';
  anchor.parentNode.insertBefore(panel,anchor);
  function render(){
   try{
    const main=Array.isArray(window.officialAgents)?window.officialAgents:(typeof officialAgents!=='undefined'?officialAgents:[]);
    const admins=Array.isArray(window.agentsAdministratifs)?window.agentsAdministratifs:(typeof agentsAdministratifs!=='undefined'?agentsAdministratifs:[]);
    const active=main.filter(a=>a&&a.id&&!a.annuaire_inactif);
    const norm=v=>(String(v||'').trim()||'Non renseignée');
    const provinces={};
    active.forEach(a=>{const province=norm(a.province_affectation||a.province);if(!provinces[province])provinces[province]={total:0,inspecteur:0,controleur:0,autre:0};provinces[province].total++;const c=String(a.categorie||'').toLowerCase();if(c.includes('inspect'))provinces[province].inspecteur++;else if(c.includes('control'))provinces[province].controleur++;else provinces[province].autre++;});
    const sorted=Object.entries(provinces).sort((a,b)=>b[1].total-a[1].total);
    const inspect=active.filter(a=>String(a.categorie||'').toLowerCase().includes('inspect')).length;
    const control=active.filter(a=>String(a.categorie||'').toLowerCase().includes('control')).length;
    const complete=active.filter(a=>a.nom_complet_transcrit&&a.matricule_transcrit&&a.province_affectation&&a.lieu_affectation).length;
    const pct=active.length?Math.round(complete/active.length*100):0;
    const cards=[['Fiches du registre principal',active.length],['Inspecteurs',inspect],['Contrôleurs',control],['Agents administratifs (liste séparée)',admins.length]];
    document.getElementById('igt-nd-cards').innerHTML=cards.map(x=>'<div class="igt-nd-card"><small>'+x[0]+'</small><strong>'+Number(x[1]).toLocaleString('fr-FR')+'</strong></div>').join('');
    const max=Math.max(1,...sorted.map(x=>x[1].total));
    document.getElementById('igt-nd-provinces').innerHTML=sorted.length?sorted.slice(0,15).map(([name,v])=>'<div class="igt-nd-row"><span>'+name.replace(/[<>]/g,'')+'</span><div class="igt-nd-track"><div class="igt-nd-fill" style="width:'+(v.total/max*100)+'%"></div></div><b>'+v.total+'</b></div>').join(''):'Aucune donnée provinciale disponible.';
    document.getElementById('igt-nd-quality').innerHTML='<div style="font-size:30px;font-weight:850;color:#12628e">'+pct+' %</div><div style="font-size:12px;color:#61788c;margin:4px 0 10px">Fiches avec nom, matricule, province et lieu renseignés</div><div class="igt-nd-track"><div class="igt-nd-fill" style="width:'+pct+'%"></div></div><p style="font-size:12px;color:#526c80;line-height:1.5">'+complete.toLocaleString('fr-FR')+' fiches complètes sur '+active.length.toLocaleString('fr-FR')+'. Les champs vides doivent être vérifiés et complétés par les services habilités.</p>';
    document.getElementById('igt-nd-breakdown').innerHTML='<table><thead><tr><th>Province</th><th>Total</th><th>Inspecteurs</th><th>Contrôleurs</th><th>Autres catégories</th></tr></thead><tbody>'+sorted.map(([n,v])=>'<tr><td>'+n.replace(/[<>]/g,'')+'</td><td>'+v.total+'</td><td>'+v.inspecteur+'</td><td>'+v.controleur+'</td><td>'+v.autre+'</td></tr>').join('')+'</tbody></table>';
    document.getElementById('igt-nd-export').onclick=function(){const rows=[['Province','Total','Inspecteurs','Contrôleurs','Autres catégories'],...sorted.map(([n,v])=>[n,v.total,v.inspecteur,v.controleur,v.autre])];const csv='\ufeff'+rows.map(r=>r.map(v=>'"'+String(v).replace(/"/g,'""')+'"').join(';')).join('\r\n');const blob=new Blob([csv],{type:'text/csv;charset=utf-8'});const url=URL.createObjectURL(blob);const a=document.createElement('a');a.href=url;a.download='ANI-IGT_synthese_nationale.csv';a.click();setTimeout(()=>URL.revokeObjectURL(url),1000);};
   }catch(err){document.getElementById('igt-nd-cards').textContent='Impossible de calculer la synthèse : '+(err&&err.message?err.message:'erreur inconnue');}
  }
  render();
 }
 if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',init,{once:true});else init();
})();
