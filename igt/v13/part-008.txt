<script id="igt-professional-final-behavior">
(function(){
 const $=id=>document.getElementById(id);
 const toggle=$('igt-header-menu-toggle'),menu=$('igt-header-dropdown');
 const profile=$('igt-auth-user'),agentTools=$('igt-agent-tools');
 if(profile&&$('igt-menu-profile'))$('igt-menu-profile').appendChild(profile);
 if(agentTools&&$('igt-menu-messages')){
   // Keep the messages and administrator actions inside the single clean header menu.
   $('igt-menu-messages').appendChild(agentTools);
   const admin=$('igt-open-admin');
   if(admin)$('igt-menu-admin').appendChild(admin);
 }
 if(toggle&&menu){
   toggle.addEventListener('click',()=>{menu.hidden=!menu.hidden;toggle.setAttribute('aria-expanded',String(!menu.hidden));});
   document.addEventListener('click',e=>{if(!menu.contains(e.target)&&!toggle.contains(e.target)){menu.hidden=true;toggle.setAttribute('aria-expanded','false');}});
   menu.addEventListener('click',e=>{if(e.target.closest('a')){menu.hidden=true;toggle.setAttribute('aria-expanded','false');}});
 }
})();
